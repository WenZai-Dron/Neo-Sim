# 游戏根目录文件结构
```
{游戏根目录}/
└── NeoSim/
    ├── Skins/                                  # 皮肤目录
    ├── Buildings/                              # 建筑目录
    └── data/                                   # 数据目录
        │
        ├── {saveName}/                         # [非服务器]按世界存档名隔离
        │   ├── data.json                       # 世界级存档数据
        │   │
        │   └── {cityName}/                     # 城市目录
        │       ├── data.json                   # 城市数据
        │       │                               #   { mode, population, dayOfWeek, day, credit,
        │       │                               #     runGuiSent, joinedPlayers }
        │       ├── player.json                 # 城市玩家列表
        │       │                               #   { "players": ["player1", "player2", ...] }
        │       └── npc/                        # NPC目录
        │           └── {npcName}.json          # 单个NPC存档文件
        │                                       #   { name, surname, givenName, sex, skin,
        │                                       #     uuid, position: {x,y,z}, yaw, pitch,
        │                                       #     health, maxHealth }
        │
        ├── data.json                           # [服务器]全局数据
        │
        └── {cityName}/                         # [服务器]
            ├── data.json
            ├── player.json
            └── npc/
                └── {npcName}.json
